```dart
// lib/core/nfc/card_uid.dart
import 'dart:typed_data';

/// Formats an NFC tag UID as uppercase hex without separators,
/// e.g. `[0x04, 0xDB, ...]` → `04DB8FE23B1590` (API contract §5.2 `card_uid`).
String formatCardUid(Uint8List bytes) {
  if (bytes.isEmpty) {
    throw ArgumentError.value(bytes, 'bytes', 'UID must not be empty');
  }
  final buffer = StringBuffer();
  for (final byte in bytes) {
    buffer.write(byte.toRadixString(16).padLeft(2, '0'));
  }
  return buffer.toString().toUpperCase();
}
```

```dart
// lib/core/nfc/card_message.dart
import 'dart:convert';
import 'dart:typed_data';

import 'package:nfc_manager/ndef_record.dart';

/// MIME type of record 1 on every wall card. Must match the
/// NDEF_DISCOVERED intent filter in AndroidManifest.xml (lowercase).
const cardMimeType = 'application/vnd.centrixplus.nfcatt';

/// Android package launched by the Android Application Record (record 2).
const appPackageName = 'com.centrixplus.nfc_attendance';

/// NDEF message written on wall cards (spec §9 step 1).
NdefMessage buildCardMessage() {
  return NdefMessage(records: [
    NdefRecord(
      typeNameFormat: TypeNameFormat.media,
      type: Uint8List.fromList(ascii.encode(cardMimeType)),
      identifier: Uint8List(0),
      payload: Uint8List.fromList(utf8.encode('v1')),
    ),
    NdefRecord(
      typeNameFormat: TypeNameFormat.external,
      type: Uint8List.fromList(ascii.encode('android.com:pkg')),
      identifier: Uint8List(0),
      payload: Uint8List.fromList(utf8.encode(appPackageName)),
    ),
  ]);
}
```

```kotlin
package com.centrixplus.nfc_attendance

import android.content.Intent
import android.nfc.NfcAdapter
import android.nfc.Tag
import android.os.Build
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel

class MainActivity : FlutterActivity() {
    private var channel: MethodChannel? = null
    private var pendingUid: String? = null

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        channel = MethodChannel(flutterEngine.dartExecutor.binaryMessenger, CHANNEL).also {
            it.setMethodCallHandler { call, result ->
                when (call.method) {
                    "takeLaunchTag" -> {
                        result.success(pendingUid)
                        pendingUid = null
                    }
                    else -> result.notImplemented()
                }
            }
        }
        pendingUid = consumeUid(intent)
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        val uid = consumeUid(intent) ?: return
        channel?.invokeMethod("launchTag", uid)
    }

    /** UID of the tag that started/resumed us, then marks the intent used so a
     *  recreated activity (e.g. rotation) does not report the same tap twice. */
    private fun consumeUid(intent: Intent?): String? {
        if (intent == null || intent.action != NfcAdapter.ACTION_NDEF_DISCOVERED) return null
        val tag: Tag? = if (Build.VERSION.SDK_INT >= 33) {
            intent.getParcelableExtra(NfcAdapter.EXTRA_TAG, Tag::class.java)
        } else {
            @Suppress("DEPRECATION")
            intent.getParcelableExtra(NfcAdapter.EXTRA_TAG)
        }
        intent.action = Intent.ACTION_MAIN
        return tag?.id?.joinToString("") { "%02X".format(it) }
    }

    companion object {
        const val CHANNEL = "nfc_attendance/launch_tag"
    }
}
```

```dart
// lib/core/nfc/launch_tag_channel.dart
import 'dart:io' show Platform;

import 'package:flutter/services.dart';

/// UID of the wall card that launched (app closed) or resumed (app in
/// background) the app on Android. iOS never delivers launch tags.
class LaunchTagChannel {
  LaunchTagChannel({MethodChannel? channel})
      : _channel = channel ?? const MethodChannel('nfc_attendance/launch_tag');

  final MethodChannel _channel;

  /// The UID that cold-started the app, once; null afterwards or on iOS.
  Future<String?> takeLaunchTag() async {
    if (!Platform.isAndroid) return null;
    return _channel.invokeMethod<String>('takeLaunchTag');
  }

  /// Called for each tap that arrives while the app is in the background.
  void listen(void Function(String uid) onUid) {
    _channel.setMethodCallHandler((call) async {
      if (call.method == 'launchTag' && call.arguments is String) {
        onUid(call.arguments as String);
      }
    });
  }
}
```

```dart
// lib/core/nfc/nfc_reader.dart
import 'dart:convert';
import 'dart:io' show Platform;

import 'package:nfc_manager/ndef_record.dart';
import 'package:nfc_manager/nfc_manager.dart';
import 'package:nfc_manager/nfc_manager_android.dart';
import 'package:nfc_manager/nfc_manager_ios.dart';

import 'card_uid.dart';

/// Thin wrapper over nfc_manager 4.2.1.
/// Android: reader mode stays on until [stop] (and while it is on, Android does
/// not dispatch the card's NDEF to the browser). The OS turns reader mode off
/// when the activity pauses, so callers restart it on resume.
/// iOS: shows the system sheet and ends after one read.
class NfcReader {
  Future<NfcAvailability> availability() =>
      NfcManager.instance.checkAvailability();

  Future<void> start({
    required Future<void> Function(NfcTag tag) onTag,
    required void Function(String message) onError,
  }) {
    return NfcManager.instance.startSession(
      pollingOptions: {NfcPollingOption.iso14443},
      alertMessageIos: 'Hold your iPhone near the card',
      onDiscovered: (tag) async {
        try {
          await onTag(tag);
        } catch (e) {
          onError(e.toString());
        }
      },
      onSessionErrorIos: (error) => onError(error.message),
    );
  }

  Future<void> stop() => NfcManager.instance.stopSession();
}

/// The tag's UID (`04DB8FE23B1590`), or null if the platform exposes none.
String? uidOf(NfcTag tag) {
  if (Platform.isAndroid) {
    final id = NfcTagAndroid.from(tag)?.id;
    return (id == null || id.isEmpty) ? null : formatCardUid(id);
  }
  if (Platform.isIOS) {
    final id = MiFareIos.from(tag)?.identifier;
    return (id == null || id.isEmpty) ? null : formatCardUid(id);
  }
  return null;
}

/// Android-only debug text: technologies and NDEF record types on the tag.
String describeTag(NfcTag tag) {
  if (!Platform.isAndroid) return '';
  final techs = NfcTagAndroid.from(tag)?.techList.map((t) => t.split('.').last).join('/') ?? '?';
  final records = NdefAndroid.from(tag)?.cachedNdefMessage?.records ?? const <NdefRecord>[];
  final types = records
      .map((r) => '${r.typeNameFormat.name}:${ascii.decode(r.type, allowInvalid: true)}')
      .join(', ');
  return 'tech=$techs ndef=[$types]';
}

/// Android-only: writes [message] to the tag and returns a result line.
Future<String> writeCard(NfcTag tag, NdefMessage message) async {
  final ndef = NdefAndroid.from(tag);
  if (ndef == null) return 'not an NDEF tag';
  if (!ndef.isWritable) return 'tag is read-only';
  if (ndef.maxSize < message.byteLength) {
    return 'too small: ${ndef.maxSize} < ${message.byteLength} bytes';
  }
  await ndef.writeNdefMessage(message);
  return 'written ${message.byteLength} bytes';
}
```

```dart
// lib/spike/spike_page.dart
import 'package:flutter/material.dart';
import 'package:geolocator/geolocator.dart';
import 'package:nfc_manager/nfc_manager.dart';

import '../core/nfc/card_message.dart';
import '../core/nfc/launch_tag_channel.dart';
import '../core/nfc/nfc_reader.dart';

/// Throw-away screen for Plan 1. Deleted in Plan 3.
class SpikePage extends StatefulWidget {
  const SpikePage({super.key});

  @override
  State<SpikePage> createState() => _SpikePageState();
}

class _SpikePageState extends State<SpikePage> with WidgetsBindingObserver {
  final _reader = NfcReader();
  final _launchTags = LaunchTagChannel();
  final _log = <String>[];
  NfcAvailability? _availability;
  bool _writeArmed = false;
  String _gps = 'no fix yet';

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _launchTags.listen((uid) => _onUid(uid, source: 'launch intent (app in background)'));
    _takeLaunchTag();
    _restartNfc();
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    _reader.stop().catchError((_) {});
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) _restartNfc();
  }

  Future<void> _takeLaunchTag() async {
    final uid = await _launchTags.takeLaunchTag();
    if (uid != null) await _onUid(uid, source: 'launch intent (app was closed)');
  }

  Future<void> _restartNfc() async {
    try {
      await _reader.stop();
    } catch (_) {
      // No session running yet.
    }
    final availability = await _reader.availability();
    if (!mounted) return;
    setState(() => _availability = availability);
    if (availability != NfcAvailability.enabled) return;
    await _reader.start(onTag: _onTag, onError: (m) => _add('NFC error: $m'));
  }

  Future<void> _onTag(NfcTag tag) async {
    final uid = uidOf(tag);
    if (_writeArmed) {
      setState(() => _writeArmed = false);
      final result = await writeCard(tag, buildCardMessage());
      _add('WRITE ${uid ?? '?'}: $result');
      return;
    }
    if (uid == null) {
      _add('tag without UID');
      return;
    }
    await _onUid(uid, source: 'reader mode (app open)', extra: describeTag(tag));
  }

  Future<void> _onUid(String uid, {required String source, String? extra}) async {
    _add('CARD $uid via $source${extra == null ? '' : ' | $extra'}');
    await _captureGps();
  }

  Future<void> _captureGps() async {
    final stopwatch = Stopwatch()..start();
    try {
      var permission = await Geolocator.checkPermission();
      if (permission == LocationPermission.denied) {
        permission = await Geolocator.requestPermission();
      }
      if (permission == LocationPermission.denied ||
          permission == LocationPermission.deniedForever) {
        _setGps('permission: ${permission.name}');
        return;
      }
      final p = await Geolocator.getCurrentPosition(
        locationSettings: const LocationSettings(
          accuracy: LocationAccuracy.high,
          timeLimit: Duration(seconds: 10),
        ),
      );
      _setGps('${p.latitude.toStringAsFixed(6)}, ${p.longitude.toStringAsFixed(6)}'
          ' ±${p.accuracy.toStringAsFixed(1)} m, mocked=${p.isMocked},'
          ' fix in ${stopwatch.elapsedMilliseconds} ms');
    } catch (e) {
      _setGps('error after ${stopwatch.elapsedMilliseconds} ms: $e');
    }
  }

  void _setGps(String text) {
    if (!mounted) return;
    setState(() => _gps = text);
    _add('GPS $text');
  }

  void _add(String line) {
    if (!mounted) return;
    final time = DateTime.now().toIso8601String().substring(11, 19);
    setState(() => _log.insert(0, '$time  $line'));
  }

  Future<void> _armWrite() async {
    final ok = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Write wall card?'),
        content: const Text(
          'The next card you tap is erased (its website link is removed) and gets '
          'our app records. NFC Tools can rewrite it later.',
        ),
        actions: [
          TextButton(onPressed: () => Navigator.pop(context, false), child: const Text('Cancel')),
          FilledButton(onPressed: () => Navigator.pop(context, true), child: const Text('Arm write')),
        ],
      ),
    );
    if (ok == true) {
      setState(() => _writeArmed = true);
      _add('write armed — tap the card');
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('NFC spike')),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          Text('NFC: ${_availability?.name ?? 'checking…'}'),
          const SizedBox(height: 8),
          Text('GPS: $_gps'),
          const SizedBox(height: 16),
          Wrap(spacing: 8, runSpacing: 8, children: [
            OutlinedButton(onPressed: _captureGps, child: const Text('Get GPS now')),
            FilledButton.tonal(
              onPressed: _writeArmed ? null : _armWrite,
              child: Text(_writeArmed ? 'Tap card to write…' : 'Write wall card'),
            ),
          ]),
          const Divider(height: 32),
          for (final line in _log)
            Padding(
              padding: const EdgeInsets.symmetric(vertical: 4),
              child: SelectableText(line, style: const TextStyle(fontFamily: 'monospace', fontSize: 12)),
            ),
        ],
      ),
    );
  }
}
```
